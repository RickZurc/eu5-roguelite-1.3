# Hidden redraw helper. Draws three new offers and re-opens the selection event.
#
# This exists so the 2566-entry random_list (rl_rand_se, inlined by rl_pick_three)
# appears ONCE in the script instead of being expanded into all ~2566 banish
# options. Firing an event by id is a cheap reference, whereas calling a
# parameterised scripted effect expands its whole body at every call site --
# which made hovering a Banish option lag the game.
rl_status.2 = {
	type = country_event
	hidden = yes
	# Hidden events never display these, but the engine still warns without them.
	title = rl_status.hidden
	desc = rl_status.hidden

	immediate = {
		rl_pick_three = yes
		trigger_event_non_silently = { id = rl_events.1 }
	}
}

# Debug helper: grant 10 reroll tokens. Fire from the console with
# "event rl_status.3" (EU5 has no console command to set a script variable).
rl_status.3 = {
	type = country_event
	hidden = yes
	title = rl_status.hidden
	desc = rl_status.hidden

	immediate = {
		change_variable = { name = rl_reroll_tokens add = 10 }
	}
}

# Token shop. Spend banked reroll tokens on lasting benefits.
rl_status.5 = {
	type = country_event
	title = rl_status.5.title
	desc = rl_status.5.desc

	outcome = neutral

	# The cleanse price grows with the curses held, so it is worked out when
	# the shop opens and stored for the option text, trigger and effect.
	immediate = {
		set_variable = { name = rl_cleanse_cost value = rl_cleanse_cost }
	}

	# Cleanse curses (2 tokens + 1 per curse stack, max 10) -- only when cursed.
	option = {
		name = rl_status.5.cleanse
		trigger = {
			var:rl_reroll_tokens >= var:rl_cleanse_cost
			OR = {
				has_country_modifier = modifier_rl_curse_1
				has_country_modifier = modifier_rl_curse_2
				has_country_modifier = modifier_rl_curse_3
				has_country_modifier = modifier_rl_curse_4
				has_country_modifier = modifier_rl_curse_5
			}
		}
		custom_tooltip = rl_status.5.cleanse.tt
		hidden_effect = {
			rl_cleanse_curses = yes
		}
	}

	# Guarantee the next roll is legendary (5 tokens).
	option = {
		name = rl_status.5.legendary
		trigger = {
			var:rl_reroll_tokens >= 5
			NOT = { var:rl_force_legendary = 1 }
		}
		custom_tooltip = rl_status.5.legendary.tt
		hidden_effect = {
			rl_buy_legendary = yes
		}
	}

	# Draw an extra selection right now (4 tokens).
	option = {
		name = rl_status.5.extra
		trigger = { var:rl_reroll_tokens >= 4 }
		custom_tooltip = rl_status.5.extra.tt
		hidden_effect = {
			rl_buy_extra_roll = yes
		}
	}

	# The devil's bargain: free, but inflicts a permanent curse.
	option = {
		name = rl_status.5.pact
		trigger = { rl_enabled_enable_cursed = yes }
		custom_tooltip = rl_status.5.pact.tt
		hidden_effect = {
			rl_dark_pact = yes
		}
	}

	option = {
		name = rl_status.5.close
	}
}

# Chaos Draft offer, fired once at the start of a new game (see
# rl_game_start_action and the end of rl_monthly_action). Either answer consumes the offer, so it
# cannot be taken again even if the event is fired from the console. The draft
# runs inside hidden_effect: previewing rl_fun_mass_roll would expand the
# 2566-entry rl_rand_grant into the tooltip.
rl_status.6 = {
	type = country_event
	title = rl_status.6.title
	desc = rl_status.6.desc

	outcome = positive

	option = {
		name = rl_status.6.go
		trigger = { has_variable = rl_chaos_draft_offered }
		custom_tooltip = rl_status.6.go.tt
		hidden_effect = {
			remove_variable = rl_chaos_draft_offered
			rl_fun_mass_roll = yes
			trigger_event_non_silently = { id = rl_status.7 }
		}
	}

	option = {
		name = rl_status.6.cancel
		hidden_effect = {
			remove_variable = rl_chaos_draft_offered
		}
	}
}
