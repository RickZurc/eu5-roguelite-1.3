# Chaos Draft (fun option) is offered ONCE, at the start of a NEW game only.
# on_game_start is a vanilla engine hook that fires for new games and never on
# a loaded save, and game rules are already set when it runs (vanilla reads
# has_game_rule there). It only leaves a marker; the offer itself is made from
# rl_monthly_action, which already shows the upgrade popups reliably. (An
# earlier version used CMF's on_game_start_after_lobby_human_country, which CMF
# fires from a GUI widget's on_start -- the popup never appeared.) The marker
# expires after a year so a player who switches country later is not offered
# it again; each country is also flagged once it has been offered.
on_game_start = {
    on_actions = {
        rl_game_start_action
    }
}

rl_game_start_action = {
    effect = {
        if = {
            limit = { rl_enabled_chaos_draft = yes }
            set_global_variable = {
                name = rl_chaos_draft_pending
                years = 1
            }
        }
    }
}

# roll for every country. Each source can be toggled in the Community Mod Menu.
on_great_battle_won = {
    on_actions = {
        rl_battle_won_action
    }
}

on_new_age = {

}

cmf_on_mod_registration = {
    on_actions = {
        rl_cmf_register_action
    }
}

rl_cmf_register_action = {
    trigger = {}
    effect = {
        rl_cmf_register = yes
    }
}

# Handle clicks on our action-bar button: open the status window.
cmf_on_callback = {
    on_actions = {
        rl_on_callback_action
    }
}

rl_on_callback_action = {
    trigger = {}
    effect = {
        if = {
            limit = { var:cmf_callback = flag:rl_status_button }
            trigger_event_non_silently = { id = rl_status.1 }
        }
    }
}

rl_monthly_action = {
    trigger = {}
    effect = {
        # Backfill any variables this country is missing. The init branch below
        # only ever runs once per country, so saves started on an older version
        # of the mod would otherwise never get variables added later -- and
        # change_variable / comparisons against an unset variable silently do
        # nothing (that is what broke rl_pacts_made and blocked artifact rolls).
        rl_ensure_vars = yes

        if = {
            limit = {
                has_variable = num_advances
            }
            # Every advance the country researches grants rl_roll_mtd = 1, so
            # modifier:rl_roll_mtd is the live count of researched advances.
            # When it grows past the last synced count, the player just gained
            # an advance, so we open a selection roll (if enabled in settings).
            if = {
                limit = {
                    modifier:rl_roll_mtd > var:num_advances
                }
                if = {
                    limit = {
                        rl_enabled_roll_on_research = yes
                    }
                    rl_do_roll = yes
                }
                # Always sync the baseline to the current count, even if rolls
                # from research are disabled, so we never build up a backlog.
                set_variable = {
                    name = num_advances
                    value = { add = modifier:rl_roll_mtd }
                }
            }
            else_if = {
                # The live count can also DROP: many advances are religion- (or
                # government-/culture-) gated via potential, so changing religion
                # invalidates them and their rl_roll_mtd stops applying. Without
                # this branch the baseline would stay too high and the player
                # would have to re-research that many advances before any further
                # selection fired. Re-sync downward so the next research rolls.
                limit = {
                    modifier:rl_roll_mtd < var:num_advances
                }
                set_variable = {
                    name = num_advances
                    value = { add = modifier:rl_roll_mtd }
                }
            }
        } else = {
            # First time we see this country. All other state is initialized
            # by rl_ensure_vars, which already ran at the top of this effect.
            rl_init_country = yes
        }

        # Chaos Draft: offer once, at a new game's first monthly pulse. Placed
        # last so rl_ensure_vars and rl_init_country have already run.
        if = {
            limit = {
                is_ai = no
                has_global_variable = rl_chaos_draft_pending
                NOT = { has_variable = rl_chaos_draft_done }
                rl_enabled_chaos_draft = yes
            }
            set_variable = rl_chaos_draft_done
            set_variable = rl_chaos_draft_offered
            trigger_event_non_silently = { id = rl_status.6 }
        }
    }
}

# Winning a great battle grants a banked reroll token to the victor's country
# (scope:actor is the winning army, its owner is the country). Rewards military
# play without flooding the player with upgrades.
rl_battle_won_action = {
    trigger = {}
    effect = {
        if = {
            limit = {
                exists = scope:actor.owner
                rl_enabled_roll_on_battle = yes
            }
            scope:actor.owner = {
                rl_grant_reroll_token = yes
            }
        }
    }
}

# A new age is a milestone: grant a roll (root is the country).
rl_new_age_action = {
    trigger = {}
    effect = {
        if = {
            limit = {
                rl_enabled_roll_on_new_age = yes
            }
            rl_do_roll_legendary = yes
        }
    }
}

# Notoriety consequences: once a country's notoriety reaches tier 3 (20+
# upgrades), each year there is a growing chance that the world reacts. The
# work happens in rl_status.9's immediate, so the event text can name who got
# the casus belli; the AI gets the same backlash without the popup.
yearly_country_pulse = {
    on_actions = {
        rl_notoriety_yearly_action
    }
}

rl_notoriety_yearly_action = {
    trigger = {
        has_variable = rl_threat
        var:rl_threat >= 20
        rl_enabled_enable_threat = yes
    }
    effect = {
        set_variable = { name = rl_backlash_roll value = 0 }
        if = {
            limit = { var:rl_threat >= 55 }
            random_list = {
                60 = { set_variable = { name = rl_backlash_roll value = 1 } }
                40 = { }
            }
        }
        else_if = {
            limit = { var:rl_threat >= 35 }
            random_list = {
                40 = { set_variable = { name = rl_backlash_roll value = 1 } }
                60 = { }
            }
        }
        else = {
            random_list = {
                25 = { set_variable = { name = rl_backlash_roll value = 1 } }
                75 = { }
            }
        }
        if = {
            limit = { var:rl_backlash_roll = 1 }
            if = {
                limit = { is_ai = no }
                trigger_event_non_silently = { id = rl_status.9 }
            }
            else = {
                trigger_event_silently = { id = rl_status.9 }
            }
        }
    }
}
