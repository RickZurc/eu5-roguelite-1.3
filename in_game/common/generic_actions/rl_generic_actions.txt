# Hand-written generic actions (NOT regenerated).

# Bribe tribesmen to settle as peasants. Shown in the Tribes estate's action
# menu (see in_game/gui/estate_actions_lateralview.gui).
#
# Each use settles 25% of every Tribes-estate tribesmen pop as peasants. The
# price is a share of the tribesmen: 4 years' worth of the income that the
# settled people represent, i.e.
#     12 months of income  x  tribesmen share of the population
# (0.25 of the tribesmen x 48 months = 12 x share). A fully tribal country pays
# a full year of income per use -- the top end of what vanilla charges for an
# action -- and every use is cheaper than the last, because fewer tribesmen
# remain. The chiefs resent losing their people, so the Tribes estate's
# satisfaction drops as well.
rl_bribe_tribesmen = {
	type = owncountry
	sound = BRIBE_ESTATE

	# Player-only: the AI has no weighting for it, and without a message type
	# in gui/messagetypes.txt (a large vanilla file we would have to replace)
	# a message would only warn.
	ai_tick = never
	automation_tick = never
	show_message = no

	potential = {
		scope:actor = {
			country_has_estate = estate_type:tribes_estate
		}
		scope:target ?= {
			estate_type = estate_type:tribes_estate
		}
	}

	allow = {
		custom_tooltip = {
			text = rl_bribe_tribesmen_has_tribesmen_tt
			scope:actor.rl_tribal_tribesmen >= 1
		}
	}

	cooldown = {
		type = rl_bribe_tribesmen_cooldown
		years = 2
	}

	price = price:rl_bribe_tribesmen
	price_modifier = {
		add = {
			desc = "rl_bribe_tribesmen_price_income"
			value = scope:actor.monthly_income_trade_and_tax
			# Never free, even for a country running at a loss.
			min = 1
			multiply = 12
		}
		multiply = {
			desc = "rl_bribe_tribesmen_price_share"
			value = scope:actor.rl_tribal_tribesmen_share
		}
	}

	select_trigger = {
		looking_for_a = estate
		source = actor
		target_flag = target
		name = "mission_select_estate"
		column = {
			data = name
		}
		enabled = {
			estate_type = estate_type:tribes_estate
		}
	}

	effect = {
		scope:actor = {
			# split_pop inside every_pop would list every single pop in the
			# tooltip, so the conversion is summarised instead.
			custom_tooltip = rl_bribe_tribesmen_effect_tt
			hidden_effect = {
				every_pop = {
					limit = {
						pop_type = pop_type:tribesmen
						estate_type = estate_type:tribes_estate
					}
					split_pop = {
						fraction = 0.25
						type = pop_type:peasants
					}
				}
			}
			add_estate_satisfaction = {
				type = estate_type:tribes_estate
				value = -0.1
			}
		}
	}
}
